cmake_minimum_required(VERSION 3.16)
project(x360_xdvdfs_tests LANGUAGES CXX)
enable_testing()
add_library(x360_xdvdfs STATIC android/app/src/main/cpp/xdvdfs.cpp)
target_include_directories(x360_xdvdfs PUBLIC android/app/src/main/cpp)
target_compile_features(x360_xdvdfs PUBLIC cxx_std_20)
target_compile_options(x360_xdvdfs PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_executable(xdvdfs_tests tests/xdvdfs_tests.cpp)
target_link_libraries(xdvdfs_tests PRIVATE x360_xdvdfs)
add_test(NAME xdvdfs_cxx_read_mount_load COMMAND xdvdfs_tests)
