cmake_minimum_required(VERSION 3.16)
project(x360_galaxy_v52_tests VERSION 5.2.0 LANGUAGES CXX)
enable_testing()
add_library(x360_core STATIC android/app/src/main/cpp/xdvdfs.cpp android/app/src/main/cpp/xex_loader.cpp android/app/src/main/cpp/xex_crypto.cpp android/app/src/main/cpp/xex_lzx.cpp)
target_include_directories(x360_core PUBLIC android/app/src/main/cpp)
target_compile_features(x360_core PUBLIC cxx_std_20)
target_compile_options(x360_core PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_executable(xdvdfs_tests tests/xdvdfs_tests.cpp)
target_link_libraries(xdvdfs_tests PRIVATE x360_core)
add_test(NAME xdvdfs_cxx_read_mount_load COMMAND xdvdfs_tests)
add_executable(xex_loader_tests tests/xex_loader_tests.cpp)
target_link_libraries(xex_loader_tests PRIVATE x360_core)
add_test(NAME xex_header_pe_mapping_and_aes COMMAND xex_loader_tests)

add_library(x360_pm4 STATIC android/app/src/main/cpp/xenos_pm4.cpp)
target_include_directories(x360_pm4 PUBLIC android/app/src/main/cpp)
target_compile_features(x360_pm4 PUBLIC cxx_std_20)
target_compile_options(x360_pm4 PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_executable(pm4_tests tests/pm4_tests.cpp)
target_link_libraries(pm4_tests PRIVATE x360_pm4)
add_test(NAME pm4_ring_decoder COMMAND pm4_tests)
