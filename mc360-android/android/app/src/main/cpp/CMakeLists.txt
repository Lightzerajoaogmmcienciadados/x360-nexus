cmake_minimum_required(VERSION 3.22.1)
project(mc360port LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
if(ANDROID_ABI STREQUAL "arm64-v8a")
  set(RUST_TARGET "aarch64-linux-android")
elseif(ANDROID_ABI STREQUAL "armeabi-v7a")
  set(RUST_TARGET "armv7-linux-androideabi")
else()
  message(FATAL_ERROR "Unsupported ABI: ${ANDROID_ABI}")
endif()
get_filename_component(ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../../../.." ABSOLUTE)
set(RUST_LIB "${ROOT}/native/rust-core/target/${RUST_TARGET}/release/libmc360_core.a")
if(NOT EXISTS "${RUST_LIB}")
  message(FATAL_ERROR "Rust core not built: ${RUST_LIB}")
endif()
add_library(rustcore STATIC IMPORTED)
set_target_properties(rustcore PROPERTIES IMPORTED_LOCATION "${RUST_LIB}")
add_library(mc360port SHARED bridge.cpp)
find_library(GLES2 GLESv2)
find_library(LOG log)
target_link_libraries(mc360port PRIVATE rustcore ${GLES2} ${LOG} dl m)
