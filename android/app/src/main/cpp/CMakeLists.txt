cmake_minimum_required(VERSION 3.22.1)
project(x360_xdvdfs_android LANGUAGES CXX)
add_library(x360xdvdfs SHARED xdvdfs.cpp xex_loader.cpp xex_crypto.cpp xex_lzx.cpp xenos_pm4.cpp xenos_video.cpp gpu_pm4_jni.cpp jni_bridge.cpp gpu_settings.cpp)
target_compile_features(x360xdvdfs PRIVATE cxx_std_20)
target_compile_options(x360xdvdfs PRIVATE -Wall -Wextra -Wpedantic)
find_library(android_log log)
if(android_log)
  target_link_libraries(x360xdvdfs PRIVATE ${android_log})
endif()

find_library(egl_lib EGL)
find_library(gles_lib GLESv3)
find_library(android_lib android)
target_link_libraries(x360xdvdfs PRIVATE ${egl_lib} ${gles_lib} ${android_lib})
